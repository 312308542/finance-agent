```python
NEWS_ENTITY_RULE_VERSION = "ashare_news_entity_v1"


@dataclass(frozen=True)
class NewsEntityDecision:
    status: Literal["passed", "failed", "ambiguous"]
    reason: str
    matched_by: str | None
    expected_exchange: str
    aliases: tuple[str, ...]


def validate_ashare_news_entity(
    *, symbol: str, asset_name: str, title: str | None, summary: str | None
) -> NewsEntityDecision:
    text = normalize_entity_text(" ".join(filter(None, [title, summary])))
    expected_exchange = ashare_exchange_suffix(symbol)
    aliases = company_name_aliases(asset_name)
    conflicting = conflicting_exchange_reference(text, symbol, expected_exchange)
    if conflicting:
        return NewsEntityDecision("failed", "conflicting_exchange_suffix", None, expected_exchange, aliases)
    if normalize_entity_text(asset_name) in text:
        return NewsEntityDecision("passed", "canonical_name", "canonical_name", expected_exchange, aliases)
    matched_alias = next((alias for alias in aliases if alias in text), None)
    if matched_alias:
        return NewsEntityDecision("passed", "company_alias", "company_alias", expected_exchange, aliases)
    if exchange_qualified_reference(text, symbol, expected_exchange):
        return NewsEntityDecision("passed", "matching_exchange_suffix", "matching_exchange_suffix", expected_exchange, aliases)
    if re.search(rf"(?<!\d){re.escape(normalize_ashare_symbol(symbol))}(?!\d)", text):
        return NewsEntityDecision("ambiguous", "bare_symbol_only", None, expected_exchange, aliases)
    return NewsEntityDecision("ambiguous", "target_entity_not_proven", None, expected_exchange, aliases)
```

```python
class StockNewsNormalizationResult:
    events: list[EventRecordData]
    evidence: list[EvidenceData]
    entity_validation: JsonDict


def fetch_stock_news(self, *, symbol: str, asset_name: str, limit: int | None = None) -> EventRecordsResult:
    clean_name = str(asset_name or "").strip()
    if not clean_name or clean_name == strip_ashare_exchange_prefix(symbol):
        return EventRecordsResult(
            provider_name=self.provider_name,
            status="unavailable",
            collected_at=datetime.now(tz=UTC),
            payload={"reason": "missing_asset_name", "symbol": symbol},
        )
    query = f"{strip_ashare_exchange_prefix(symbol)} {clean_name}"
    df = ak.stock_news_em(symbol=query)
    normalized = normalize_ashare_stock_news(
        df,

        asset_name=clean_name,
        source="akshare:stock_news_em",
        collected_at=collected_at,
        limit=limit,
    )
```

```python
def list_stock_news_for_entity_revalidation(
    self, *, limit: int | None = None
) -> list[EventRecordORM]:
    statement = (
        select(EventRecordORM)
        .where(EventRecordORM.source == STOCK_NEWS_SOURCE)
        .order_by(EventRecordORM.collected_at, EventRecordORM.event_id)
    )
    if limit is not None:
        statement = statement.limit(max(limit, 1))
    return list(self.session.scalars(statement))


def update_news_entity_validations(
    self, rows: Sequence[JsonDict], *, chunk_size: int = 500
) -> int:

        self.session,

        rows=rows,
        chunk_size=chunk_size,
    )


def update_evidence_entity_validations(
```
